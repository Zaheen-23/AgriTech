// Helper to generate dynamic pest alerts based on weather
function generateWeatherAlerts(now, days) {
  const alerts = [];
  const next3 = days.slice(0, 3);

  if (now.humidity >= 75 && now.temp >= 20 && now.temp <= 32) {
    alerts.push({
      type: 'Pest Warning',
      severity: 'high',
      icon: '🐛',
      title: 'Fungal Blight Risk High',
      desc: 'High humidity and warm climate favor fungal rust & late blight. Inspect crop undersides and consider organic neem spray or recommended fungicide.'
    });
  }
  if (now.humidity < 40 && now.temp >= 33) {
    alerts.push({
      type: 'Insect Warning',
      severity: 'medium',
      icon: '🦗',
      title: 'Aphids & Mite Surge Alert',
      desc: 'Hot and dry spell detected. Sucking pests like spider mites and aphids multiply rapidly. Ensure light irrigation.'
    });
  }
  if (next3.some(d => d.rain >= 65)) {
    alerts.push({
      type: 'Precipitation Alert',
      severity: 'medium',
      icon: '🌧️',
      title: 'Heavy Rainfall Expected',
      desc: 'Heavy rain probability exceeds 65% in next 72 hrs. Hold back on foliar pesticide spraying & urea top-dressing. Clear drainage channels.'
    });
  }
  if (days.some(d => d.max >= 37)) {
    alerts.push({
      type: 'Heatwave',
      severity: 'high',
      icon: '🔥',
      title: 'Heat Stress Danger',
      desc: 'Temperatures above 37°C predicted. Irrigate crops during early morning or evening hours to protect crop blossoms.'
    });
  }
  if (now.wind >= 22) {
    alerts.push({
      type: 'Wind Advisory',
      severity: 'low',
      icon: '💨',
      title: 'Strong Wind Drift',
      desc: 'Wind speed over 20 km/h. Avoid high-pressure chemical spraying today to prevent chemical drift to neighboring fields.'
    });
  }

  return alerts;
}

                <p className="text-slate-400 text-sm py-12 text-center">{t('cartEmpty')}</p>
              ) : (
                <div className="divide-y divide-slate-100 my-4 max-h-[60vh] overflow-y-auto">
                  {cart.map((c) => (
                    <div key={c.id} className="py-3 flex items-center justify-between">
                      <div>
                        <p className="text-sm font-bold text-slate-800">{c.name}</p>
                        <p className="text-xs text-slate-400">
                          ₹{c.price} × {c.qty} {c.unit}
                        </p>
                      </div>
                      <div className="flex items-center gap-2">
                        <span className="font-bold text-sm text-emerald-700">₹{c.price * c.qty}</span>
                        <button
                          onClick={() => removeFromCart(c.id)}
                          className="text-rose-500 hover:text-rose-700 p-1"
                        >
                          <Trash2 className="w-4 h-4" />
                        </button>
                      </div>
                    </div>
                  ))}
                </div>
              )}
            </div>

            {cart.length > 0 && (
              <div className="border-t pt-4 space-y-3">
                <div className="flex justify-between text-base font-bold text-slate-900">
                  <span>{t('total')}</span>
                  <span>₹{totalCartPrice}</span>
                </div>
                <button
                  onClick={handleCheckout}
                  className="w-full py-3 bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl font-bold transition shadow"
                >
                  {t('checkout')}
                </button>
              </div>
            )}
          </div>
        </div>
      )}

      {/* 2. Submit Soil Sample Modal */}
      {showSoilModal && (
        <div className="fixed inset-0 z-50 bg-black/60 backdrop-blur-xs flex items-center justify-center p-4">
          <div className="bg-white rounded-2xl max-w-md w-full p-6 shadow-2xl">
            <h3 className="text-lg font-bold text-slate-900 mb-4">{t('submitSoilSample')}</h3>
            <form onSubmit={handleSoilSubmit} className="space-y-3 text-xs">
              <div>
                <label className="block text-slate-600 font-semibold mb-1">{t('farmLocation')}</label>
                <input
                  type="text"
                  required
                  placeholder="e.g. Plot 4B, Karnal"
                  value={newSoilForm.location}
                  onChange={(e) => setNewSoilForm({ ...newSoilForm, location: e.target.value })}
                  className="w-full p-2 border border-slate-200 rounded-lg"
                />
              </div>

              <div className="grid grid-cols-2 gap-2">
                <div>
                  <label className="block text-slate-600 font-semibold mb-1">{t('phLevel')}</label>
                  <input
                    type="number"
                    step="0.1"
                    value={newSoilForm.ph}
                    onChange={(e) => setNewSoilForm({ ...newSoilForm, ph: e.target.value })}
                    className="w-full p-2 border border-slate-200 rounded-lg"
                  />
                </div>
                <div>
                  <label className="block text-slate-600 font-semibold mb-1">{t('nitrogen')}</label>
                  <input
                    type="text"
                    placeholder="e.g. 210"
                    value={newSoilForm.n}
                    onChange={(e) => setNewSoilForm({ ...newSoilForm, n: e.target.value })}
                    className="w-full p-2 border border-slate-200 rounded-lg"
                  />
                </div>
              </div>

              <div className="grid grid-cols-2 gap-2">
                <div>
                  <label className="block text-slate-600 font-semibold mb-1">{t('phosphorus')}</label>
                  <input
                    type="text"
                    placeholder="e.g. 18"
                    value={newSoilForm.p}
                    onChange={(e) => setNewSoilForm({ ...newSoilForm, p: e.target.value })}
                    className="w-full p-2 border border-slate-200 rounded-lg"
                  />
                </div>
                <div>
                  <label className="block text-slate-600 font-semibold mb-1">{t('potassium')}</label>
                  <input
                    type="text"
                    placeholder="e.g. 300"
                    value={newSoilForm.k}
                    onChange={(e) => setNewSoilForm({ ...newSoilForm, k: e.target.value })}
                    className="w-full p-2 border border-slate-200 rounded-lg"
                  />
                </div>
              </div>

              <div className="pt-4 flex gap-2">
                <button
                  type="button"
                  onClick={() => setShowSoilModal(false)}
                  className="flex-1 py-2 border border-slate-200 rounded-lg font-bold text-slate-600"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="flex-1 py-2 bg-amber-600 hover:bg-amber-700 text-white rounded-lg font-bold"
                >
                  Submit
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* 3. Seller Contact Modal */}
      {selectedSellerModal && (
        <div className="fixed inset-0 z-50 bg-black/60 backdrop-blur-xs flex items-center justify-center p-4">
          <div className="bg-white rounded-2xl max-w-sm w-full p-6 shadow-2xl text-center">
            <span className="text-4xl">{selectedSellerModal.image}</span>
            <h3 className="text-lg font-bold text-slate-900 mt-2">{selectedSellerModal.name}</h3>
            <p className="text-xs text-slate-400">{selectedSellerModal.location}</p>

            <div className="my-4 bg-slate-50 p-4 rounded-xl text-left text-xs space-y-2">
              <p className="flex items-center gap-2">
                <User className="w-4 h-4 text-emerald-600" />
                <span className="font-bold">{selectedSellerModal.farmerName}</span>
              </p>
              <p className="flex items-center gap-2">
                <Phone className="w-4 h-4 text-emerald-600" />
                <span>{selectedSellerModal.contact}</span>
              </p>
            </div>

            <button
              onClick={() => setSelectedSellerModal(null)}
              className="w-full py-2 bg-slate-800 text-white rounded-xl text-xs font-bold"
            >
              Close
            </button>
          </div>
        </div>
      )}

      {/* ==========================================
          FOOTER
          ========================================== */}
      <footer className="bg-emerald-950 text-emerald-400 text-xs py-6 border-t border-emerald-900 mt-auto">
        <div className="max-w-7xl mx-auto px-4 text-center space-y-2">
          <p>© 2026 AgriTech Multilingual Portal. Supporting Indian Farmers across 8 Languages.</p>
          <p className="text-emerald-600">Built with React.js & Tailwind CSS</p>
        </div>
      </footer>

    </div>
  );
}
